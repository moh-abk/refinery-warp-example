2. Re-IP each host from the iLO console (BM02–BM07 first, BM01 last)

$if  = "vEthernet (OC-Servers)"
$new = "172.21.0.1X"          # set per host
$len = <prefix>; $gw = "<gateway>"   # same as now

Get-NetIPAddress -InterfaceAlias $if -AddressFamily IPv4 | Remove-NetIPAddress -Confirm:$false
Remove-NetRoute -InterfaceAlias $if -DestinationPrefix 0.0.0.0/0 -Confirm:$false -ErrorAction SilentlyContinue
New-NetIPAddress -InterfaceAlias $if -IPAddress $new -PrefixLength $len -DefaultGateway $gw
Get-VMNetworkAdapterVlan -ManagementOS   # should show Access 201
